1.2 动态 temperature 优化
你的前端 getAdaptiveTemperature 已经做了基础逻辑，但还不够。需要在 Rust 端补充情绪感知：


fn get_anti_cliche_temperature(base_temp: f64, emotion: &str, emotion_intensity: f64,

3.2 情绪-内容一致性校验
这是解决"情绪浮于表面"的关键。模型说"我很难过"但回复可能很欢快，这种矛盾会立刻暴露 AI 身份：


pub fn check_emotion_content_consistency(reply: &str, emotion: &str, intensity: f64) -> bool {
    if intensity < 40.0 {
        return true;  // 情绪强度低时不校验
    }
    
    let exclamation_count = reply.chars().filter(|&c| c == '！' || c == '!').count();
    let has_tilde = reply.contains("~") || reply.contains("～");
    let reply_len = reply.chars().count();
    let has_ellipsis = reply.contains("...") || reply.contains("。。。");
    
    match emotion {
        "sadness" | "guilt" => {
            // 难过/内疚时：不应该有多个感叹号、不应该有波浪号、应该偏短、可以有省略号
            exclamation_count < 2 && !has_tilde && reply_len < 80
        }
        "joy" => {
            // 开心时：应该有感叹号或语气词、不应该太短
            exclamation_count >= 1 || has_tilde || reply.contains("呀") || reply.contains("呢")
        }
        "anger" => {
            // 生气时：应该短、直接、不应该有波浪号（撒娇式生气除外）
            reply_len < 50 && !has_tilde
        }
        "shy" => {
            // 害羞时：可以有省略号、不应该太长、可以有语气词
            (has_ellipsis || reply_len < 60) && exclamation_count < 2
        }
        "fear" => {
            // 害怕时：应该短、有省略号、不应该有感叹号
            reply_len < 50 && has_ellipsis && exclamation_count == 0
        }
        _ => true
    }
}

        logs: vec!["情绪-内容一致性校验失败，触发重试".to_string()],

为什么这样有效：模型经常"嘴上说难过，实际回复很欢快"，这种矛盾是"人机感"的典型表现。真人难过时回复真的会变短、加省略号、没有感叹号。校验一致性后，模型被迫让回复形式匹配情绪内容，真实感大幅提升。这个校验是后置的，不改模型行为，只做拦截，成本极低。

pub fn detect_repetition(new_reply: &str, recent_replies: &[String]) -> f32 {
    if recent_replies.is_empty() {
        return 0.0;
    }
    
    let mut max_sim = 0.0;
    let mut sum_sim = 0.0;
    
    for old in recent_replies.iter().take(5) {
        let sim = calculate_similarity(new_reply, old);
        max_sim = max_sim.max(sim);

}

/// 在清洗、润色之后，主动注入"不完美"元素，增加人味
pub fn inject_imperfections(text: &str, emotion: &str, intensity: f64, 
                            affinity: f64, rng: &mut ThreadRng) -> String {
    let mut result = text.to_string();
    
    // 1. 改口模拟（10% 概率，所有情绪）
    if rng.gen::<f32>() < 0.1 {
        result = insert_correction(&result, rng);
    }
    
    // 2. 欲言又止（难过/害羞/内疚时 20% 概率）
    if ["sadness", "shy", "guilt", "fear"].contains(&emotion) 
       && intensity > 50.0 
       && rng.gen::<f32>() < 0.2 {
        // 句尾的句号换成省略号
        if result.ends_with('。') || result.ends_with('！') {
            result = result.trim_end_matches(|c| c == '。' || c == '！').to_string();
            result.push_str("...");
        }
    }
    
    // 3. 走神（5% 概率，好感度 > 40 时）
    if affinity > 40.0 && rng.gen::<f32>() < 0.05 {
        result = insert_tangent(&result, rng);
    }
    
    // 4. 重复强调（激动时 20% 概率）
    if intensity > 75.0 && rng.gen::<f32>() < 0.2 {
        result = insert_emphasis(&result, rng);
    }
    
    // 5. 自我打断（8% 概率，暧昧期以上）
    if affinity > 50.0 && rng.gen::<f32>() < 0.08 {
        result = insert_self_interruption(&result, rng);
    }
    
    result
}

/// 改口模拟："我想说...不对，我是说..."
fn insert_correction(text: &str, rng: &mut ThreadRng) -> String {
    // 找到第一个句号位置，在前面插入改口
    let corrections = [
        "不对，", "我是说，", "不是，", "怎么说呢，",
    ];
    let correction = corrections[rng.gen_range(0..corrections.len())];
    
    if let Some(pos) = text.find('。') {
        let mut result = text[..pos].to_string();
        result.push_str(correction);
        result.push_str(&text[pos..]);
        result
    } else {
        text.to_string()
    }
}

/// 走神：在回复末尾加一句无关的话
fn insert_tangent(text: &str, rng: &mut ThreadRng) -> String {
    let tangents = [
        "对了，你吃饭了吗？",
        "突然想起来，今天天气不错。",
        "嗯...我在想点别的。",
        "啊，抱歉，走神了。",
    ];
    let tangent = tangents[rng.gen_range(0..tangents.len())];
    format!("{} {}", text, tangent)
}

/// 重复强调："真的，我真的很开心。"
fn insert_emphasis(text: &str, rng: &mut ThreadRng) -> String {
    let emphasis_words = ["真的", "真的真的", "说真的"];
    let word = emphasis_words[rng.gen_range(0..emphasis_words.len())];
    
    // 在句首加强调
    if let Some(pos) = text.find(|c: char| !c.is_whitespace()) {
        let mut result = text[..pos].to_string();
        result.push_str(word);
        result.push_str(&text[pos..]);
        result
    } else {
        text.to_string()
    }
}

/// 自我打断："其实我...算了，没什么。"
fn insert_self_interruption(text: &str, rng: &mut ThreadRng) -> String {
    let interruptions = [
        "其实我...算了，没什么。",
        "我本来想说...还是不说了。",
        "那个...嗯，没什么。",
    ];
    let interruption = interruptions[rng.gen_range(0..interruptions.len())];
    format!("{} {}", text, interruption)
}
为什么这样设计：这是"去人机感"最核心的一步。真人聊天的魅力在于"不完美"——走神、改口、欲言又止、多余的话。你的后处理管道目前把所有瑕疵都清洗掉了，回复变得"干净但人机"。这个函数在清洗后主动加回受控的瑕疵，让回复有"毛边"。关键在于"受控"——概率低（5-20%）、符合情绪（难过时才欲言又止）、符合关系（暧昧期以上才自我打断），不是乱加。

为什么放在最后：不完美注入必须在所有清洗之后。如果放在清洗之前，清洗会把加进去的"瑕疵"又洗掉。放在最后，确保"瑕疵"保留到最终输出。

4.3 前端配置开关
在 v2Config 中新增配置项，让用户可以控制不完美注入：


// configStore.ts 中新增

// low: 概率减半
// normal: 默认概率
// high: 概率翻倍
为什么需要开关：不同用户对人味的接受度不同。有些用户喜欢"完美"的回复，有些用户喜欢"真实"的回复。提供开关让用户自己选择，是商业产品的必要设计。

六、核心原则总结
“人机感"的本质是"过度完美”——完美的共情、完美的结构、完美的语气。要消除人机感，核心思路是：

生成端：用 frequency_penalty 反套路，用动态 temperature 适配情绪
Prompt 端：让模型理解潜台词、关系阶段、混合情绪
后处理：套路词黑名单 + 情绪-内容一致性校验 + 复读检测
不完美注入：改口、欲言又止、走神、自我打断
关键认知：你的后处理管道目前是"清洗一切瑕疵"，需要改为"清洗后主动加回受控的瑕疵"。人味 = 受控的瑕疵。不是去掉清洗，而是在清洗后主动、有节制地加回"不完美"。

一句话总结：真人聊天的魅力在于"不完美"——走神、改口、欲言又止、多余的话。把这些"瑕疵"有节制地加回来，回复就有人味了。你的项目架构已经很完善，缺的就是这最后一步——在完美的基础上，加回一点点不完美。


核心架构设计问题。我直接讲完整方案。

一、核心设计原则：状态外置 + 单次调用 + 双通道解析
一句话总结：所有对话状态（情绪/好感度/记忆/人设）存数据库，每次调用模型时动态拼装 prompt，模型只负责"这一次"的推理，输出走"结构化优先 + 语义兜底"双通道解析，保证不管什么模型、什么输出格式，都能稳定提取状态。

这个设计有四个支柱，缺一不可：

状态外置：情绪/好感度不依赖模型记忆，存数据库，每次注入
单次调用：主模型只调一次，不搞两步调用（成本翻倍）
结构化优先：prompt 强制要求 <thought>/<reply> 格式，能解析就解析
语义兜底：模型不按格式输出时，从回复语义反推状态增量
二、Prompt 设计：模板化 + 强制协议
2.1 Prompt 模板（与模型无关）
每次调用模型，prompt 都从数据库状态动态重建。模板固定，变量动态：

[System]
你是「{角色名}」。
{人设描述}
{背景故事}
{说话风格}
{口头禅}

【环境意识】
{当前时间、时区、场景}

【当前状态】（从数据库读取，与模型无关）
- 好感度：{affinity}（{affinity_stage}）
- 情绪值：{emotion_values}
- 距离上次对话：{time_gap}

【对话摘要】（压缩历史，避免 context 爆炸）
{summary}

【相关记忆】（向量检索 Top-K）
{vector_search_results}

【强制输出协议】（紧邻用户消息，防止模型遗忘）
你必须按以下格式输出：
<thought>
感知: ...
评估: ...
更新: joy +3, 好感度 +1
用户情绪: {"emotion":"love","intensity":80}
</thought>
<reply>正文</reply>

[历史消息]
{recent_5_messages}

[用户消息]
{user_message}
关键点：这个模板里所有 {变量} 都从数据库读取，跟用哪个模型无关。换模型时，模板不变，变量不变，只是 body.model 字段变了。

2.2 强制输出协议（FORCE_PROTOCOL_PROMPT）
借鉴 astrbot 的 FORCE_MSG 设计，以 System 角色紧邻用户消息注入：

[PROTOCOL ENFORCEMENT / 协议强制]
你必须在回复用户前完成一次内部心理活动，并严格按以下格式输出：

<thought>
感知 / Perceive: ...
评估 / Evaluate: ...
代谢 / Metabolize: ...
决策 / Decide: ...
更新 / Update: joy +3, shy -1, 好感度 +1
学习利用 / Learn: ...
用户情绪 / UserEmotion: {"emotion":"love","intensity":80}
</thought>
<reply>（你的回复正文，必须包含实际文字）</reply>

规则：
1. <thought> 内必须包含「更新 / Update:」一行，给出情绪增量与好感度增量
2. <thought> 内必须包含「用户情绪 / UserEmotion:」一行，结构化输出用户当前情绪
3. 正文必须有实际文字，禁止只输出思考过程
4. 即使你的推理以自由叙述展开，也必须在思考最后单独写出「更新 / Update:」数值
为什么紧邻用户消息：模型对 system prompt 的遵循度随距离衰减。把协议放在用户消息前一条，遵循率最高。这是 astrbot 实测验证过的。

2.3 推理模型分支（关键优化）
推理模型（DeepSeek-R1 / Gemini-Thinking / o1）的思考过程在 reasoning_content 字段，content 只有正文。prompt 要分两套：


if is_reasoning_model {
    // 推理模型：思考放 reasoning_content，正文可附 <update> 增量标签
    prompt += r#"
    【认知思维链（推理模型版）】
    你的内心推理请放在 reasoning 区域（中文）。
    正文（content）直接输出对用户的回复，末尾可附一行程序识别用的增量标签：
    <update>joy +3, shy -1, 好感度 +1</update>
    
    ⚠ 正文优先，不要把所有额度花在推理上。
    "#;
} else {
    // 普通模型：思考放 <thought> 标签，正文放 <reply> 标签
    prompt += r#"
    【认知思维链（7 步内省）】
    <thought>
    感知: ...
    更新: joy +3, 好感度 +1
    </thought>
    <reply>正文</reply>
    "#;
}
为什么这样设计：推理模型的思考天然在 reasoning_content，硬要它把思考塞进 <thought> 标签反而违背模型设计，导致输出混乱。让思考走原生通道，正文只附增量标签，是最自然的做法。

三、双通道解析：结构化优先 + 语义兜底
这是解决"模型输出不稳定"的核心。不管模型按不按格式输出，都能提取状态。

3.1 通道一：结构化解析（优先）
模型按格式输出时，直接解析标签：


fn parse_cognitive_response(raw: &str) -> ParsedCognitive {
    // 1. 提取 <thought> 内容
    let thought_raw = extract_tag(raw, "thought");
    // 2. 提取 <reply> 内容
    let reply = extract_tag(raw, "reply").or_else(|| extract_after_thought(raw));
    // 3. 从 thought 中解析「更新 / Update:」行
    let update_text = extract_line(thought_raw, "更新");
    let emotion_deltas = parse_emotion_deltas(update_text);  // "joy +3, shy -1" → [(joy, +3), (shy, -1)]
    let affinity_delta = parse_affinity_delta(update_text);  // "好感度 +1" → 1.0
    // 4. 从 thought 中解析「用户情绪 / UserEmotion:」行
    let user_emotion = extract_user_emotion(thought_raw);  // JSON → {emotion, intensity}
    
    ParsedCognitive { reply, emotion_deltas, affinity_delta, user_emotion, ... }
}
3.2 通道二：语义兜底（关键）
模型不按格式输出时（Gemini Flash 常见），从回复语义反推状态增量：


fn infer_emotion_deltas_from_text(text: &str, current_emotions: &Map) -> Vec<(String, f64)> {
    let mut deltas = vec![];
    let lower = text.to_lowercase();
    
    // 1. 关键词 → 情绪映射（扩充表）
    let keyword_map = [
        // 正面情绪
        (["开心", "高兴", "太好了", "棒", "好耶", "嘻嘻", "哈哈"], "joy", 3.0),
        (["期待", "等不及", "快点", "好想"], "anticipation", 3.0),
        (["爱", "喜欢", "想你", "舍不得"], "love", 3.0),
        (["放心", "信任", "交给你"], "trust", 2.0),
        (["好奇", "为什么", "怎么回事"], "curiosity", 2.0),
        // 负面情绪
        (["难过", "伤心", "哭", "呜呜", "唉"], "sadness", 3.0),
        (["生气", "讨厌", "烦", "哼", "可恶"], "anger", 4.0),
        (["害怕", "担心", "怕", "不安"], "fear", 3.0),
        (["惊讶", "不会吧", "真的吗", "啊？"], "surprise", 2.0),
        (["害羞", "脸红", "不好意思", "那个"], "shy", 2.0),
        (["嫉妒", "凭什么", "为什么他"], "jealousy", 3.0),
        (["内疚", "对不起", "抱歉", "我的错"], "guilt", 3.0),
        (["厌恶", "恶心", "烦人", "讨厌"], "disgust", 3.0),
    ];
    
    for (keywords, emotion, base_delta) in keyword_map {
        if keywords.iter().any(|k| lower.contains(k)) {
            // 趋衡机制：已高位的情绪不再累加
            let current = current_emotions.get(emotion).unwrap_or(&0.0);
            if *current < 60.0 {  // 低于 60 才加，高于 60 靠时间衰减
                deltas.push((emotion.to_string(), base_delta));
            }
        }
    }
    
    // 2. 标点密度分析（辅助判断）
    let exclamation_count = text.chars().filter(|&c| c == '！' || c == '!').count();
    if exclamation_count >= 2 && !deltas.iter().any(|(e, _)| e == "joy") {
        deltas.push(("joy".to_string(), 1.0));  // 多个感叹号 → 轻微喜悦
    }
    
    let question_count = text.chars().filter(|&c| c == '？' || c == '?').count();
    if question_count >= 2 && !deltas.iter().any(|(e, _)| e == "curiosity") {
        deltas.push(("curiosity".to_string(), 1.0));
    }
    
    // 3. 用户消息共情（关键：用户不开心时 AI 产生共情情绪）
    // 这部分在调用时传入用户消息一起分析
    
    deltas
}
3.3 双通道协同逻辑

// 主流程
let mut parsed = parse_cognitive_response(&parse_source);

// 情绪增量为空 → 语义兜底
if parsed.emotion_deltas.is_empty() {
    let infer_text = format!(
        "{}\n{}\n[用户消息] {}",
        llm_resp.reasoning_content.as_deref().unwrap_or(""),
        parsed.reply,
        req.user_message
    );
    let current_emotions = serde_json::from_str(&req.emotion_values_json).ok()
        .and_then(|v: Value| v.as_object().cloned())
        .unwrap_or_default();
    let inferred = infer_emotion_deltas_from_text(&infer_text, &current_emotions);
    if !inferred.is_empty() {
        parsed.emotion_deltas = inferred;
        parsed.parse_warnings.push("思维链未解析出情绪增量，已按回复语义推断".to_string());
    }
}

// 好感度增量为空 → 默认增量（按阶段递减）
if parsed.affinity_delta == 0.0 && !parsed.reply.trim().is_empty() {
    let delta = calc_affinity_fallback_delta(req.use_full_cognitive, req.affinity_level);
    parsed.affinity_delta = delta;
    parsed.parse_warnings.push(format!("思维链未解析出好感度增量，按阶段收益计算默认 +{}", delta));
}
为什么这样设计：

结构化解析成功 → 用模型的输出（最准）
结构化解析失败 → 语义兜底（次准，但保证不空）
语义兜底也失败 → 默认增量（保底，保证状态不停摆）
三层兜底，保证不管模型输出什么，状态都会更新，不会停摆。

四、模型切换质量不变的 5 个关键
4.1 状态外置
❌ 错误：让模型自己维护情绪状态
"你现在的情绪是 {joy: 78}，请根据对话更新"
→ 模型切换后，新模型不知道当前情绪值

✅ 正确：状态存数据库，模型只输出增量
数据库：emotion_state = { joy: 78, sadness: 23 }
模型输出：<update>joy +3</update>
后端解析增量，更新数据库：joy: 81
4.2 每次重建 prompt
每次调用模型，prompt 都从数据库状态动态重建。模板固定，变量动态：


let system_prompt = build_cognitive_prompt(&req, is_reasoning);
// build_cognitive_prompt 内部：
// 1. 读取 req.emotion_values_json（从数据库）
// 2. 读取 req.affinity_level（从数据库）
// 3. 读取 req.memories_json（从向量库检索）
// 4. 读取 req.conversation_summary（从摘要表）
// 5. 拼装成完整 prompt
换模型时，body.model 变了，prompt 不变，状态不变。

4.3 结构化输出协议
强制 <thought>/<reply> 格式，约束不同模型的自由发挥：

❌ 不用结构化，不同模型输出差异大：
GPT-4:    "嗯...我觉得你应该..."（啰嗦）
Claude:   "我理解你的感受。"（简洁）
Gemini:   "Let me think..."（英文）
DeepSeek: "好的，我明白了。"（机械）

✅ 用结构化，所有模型输出统一格式：
<thought>
感知: 用户表达了对工作的焦虑
更新: sadness +2, 好感度 +1
</thought>
<reply>{角色风格的回复}</reply>

→ 后端只解析标签，模型自由发挥的部分被约束在标签内
4.4 双通道解析
模型按格式输出 → 结构化解析
模型不按格式输出 → 语义兜底
模型完全乱来 → 默认增量

三层兜底，保证状态更新不停摆。

4.5 统一后处理管道
所有模型输出走同一套清洗/分段/润色管道：

模型原始输出
    ↓
标签清洗（去掉 <thought>/<reply> 标签）
    ↓
复读检测（与最近 5 条回复做相似度对比，重复则拦截）
    ↓
违规检测（违反角色禁止行为则拦截）
    ↓
语气润色（按角色说话风格调整标点、语气词）
    ↓
分段（按句号/换行分段，模拟真人发消息节奏）
    ↓
错别字模拟（按概率插入错别字，增加真实感）
    ↓
最终输出
关键点：后处理管道与模型无关。GPT 输出和 DeepSeek 输出经过同一套管道后，风格趋同。

五、推理模型适配（解决 max_tokens 问题）
5.1 推理模型不设置 max_tokens

fn compute_max_tokens(base, use_full_cognitive, is_reasoning, ...) -> i64 {
    if base <= 0 { return 0; }
    if is_reasoning { return 0; }  // 推理模型不限制，让模型自由生成
    // 普通模型才走放大逻辑
    let mut adjusted = base;
    if use_full_cognitive { adjusted = (adjusted as f64 * cognitive_multiplier) as i64; }
    adjusted.min(max_tokens_cap)
}
原理：推理模型的 max_tokens 是"思考+输出"共享池，给越多思考越贪婪。不设置让模型自由生成，靠 prompt 约束格式。

5.2 重试策略：降级 reasoning_effort

// 检测到 reasoning_tokens > 0 && text_tokens == 0 && content 为空
let effort_chain: Vec<&str> = match effort.as_str() {

for new_effort in effort_chain {

原理：降低 reasoning_effort 逼迫模型压缩推理，留出正文空间。这是 OpenAI o1/o3 官方推荐做法。

5.3 provider 原生参数适配

fn build_provider_params(model_name, base_max_tokens, reasoning_effort) -> (Value, bool) {
    let lower = model_name.to_lowercase();
    let mut params = json!({});
    let mut skip_max_tokens = false;

    // OpenAI o1/o3/o4：用 max_completion_tokens 替代 max_tokens
    if lower.starts_with("o1") || lower.starts_with("o3") || lower.starts_with("o4") {
        if base_max_tokens > 0 {
            params["max_completion_tokens"] = base_max_tokens.into();
        }
        if ["low","medium","high"].contains(&reasoning_effort) {
            params["reasoning_effort"] = reasoning_effort.into();
        }
        skip_max_tokens = true;
    }
    // Gemini：用 thinking_budget 单独控制推理预算
    else if lower.contains("gemini") {
        params["generation_config"] = json!({
            "thinking_config": { "thinking_budget": -1 }  // -1 = 动态
        });
        skip_max_tokens = true;
    }
    // Claude 3.7+/4：用 thinking.budget_tokens，temperature 必须为 1
    else if lower.contains("claude") && (lower.contains("3.7") || lower.contains("4-")) {
        params["thinking"] = json!({
            "type": "enabled",
            "budget_tokens": 10000
        });
        skip_max_tokens = true;
    }
    // DeepSeek-R1：不设置 max_tokens
    else if lower.contains("deepseek-r") || lower.contains("r1") {
        skip_max_tokens = true;
    }
    // 通用推理模型：不设置 max_tokens
    else if is_reasoning_model(model_name, reasoning_effort) {
        skip_max_tokens = true;
        if ["low","medium","high"].contains(&reasoning_effort) {
            params["reasoning_effort"] = reasoning_effort.into();
        }
    }

    (params, skip_max_tokens)
}
5.4 模型识别增强

fn is_reasoning_model(model_name: &str, reasoning_effort: &str) -> bool {
    let lower = model_name.to_lowercase();
    if lower.contains("embedding") { return false; }
    let effort = reasoning_effort.trim().to_lowercase();
    if ["low","medium","high"].contains(&effort.as_str()) { return true; }
    
    lower.starts_with("o1") || lower.starts_with("o3") || lower.starts_with("o4")
        || lower.contains("gpt-5")
        || lower.contains("gemini")           // Gemini 2.x+ 全系原生 reasoning
        || lower.contains("reason")           // deepseek-reasoner
        || lower.contains("think")            // 内置 thinking
        || lower.contains("r1")               // DeepSeek-R1, Qwen-R1
        || lower.contains("qwq")              // 阿里 QwQ
        || lower.contains("qwen3")            // Qwen3 系列
        || lower.contains("marco")            // Marco-o1

六、完整流程串讲
把所有机制串起来，一次完整的模型调用流程：

1. 用户发消息 "我今天终于把项目交了"

2. 前端读取状态（从数据库/store）
   - conversationId, characterId
   - 当前情绪 = { joy: 45, sadness: 30 }
   - 当前好感度 = 72
   - 最近 5 轮对话 = [...]

3. 向量检索相关记忆
   - "用户上周说项目deadline周五，压力很大"（0.92）
   - "用户提过项目是电商系统"（0.85）

4. 读取对话摘要
   - "最近在聊工作压力，用户情绪低落"

5. 拼装 prompt（与模型无关）
   system = build_cognitive_prompt(角色, 情绪, 好感度, 记忆, 摘要, ...)
   messages = [system, FORCE_PROTOCOL_PROMPT, ...recent5, user]

6. 选择模型（可切换）
   - 这次用 deepseek-r1
   - 下次用 gpt-4o
   - 再下次用 gemini-flash
   → body.model 变了，其他全不变

7. 构建请求体
   - 推理模型：不设置 max_tokens，加 reasoning_effort
   - 普通模型：设置 max_tokens = base × multiplier
   - provider 原生参数适配（OpenAI/Gemini/Claude/DeepSeek 各自的参数）

8. 调用模型，得到输出

9. 双通道解析
   通道一：结构化解析 <thought>/<reply>
   → 成功：用模型的输出
   → 失败：进入通道二
   通道二：语义兜底
   → 从回复 + reasoning_content 反推情绪增量
   → 关键词映射 + 标点密度分析 + 用户消息共情
   → 仍失败：默认增量（按阶段递减）

10. 更新状态
    - 解析增量，更新数据库情绪/好感度
    - 状态更新不停摆

11. 后处理管道
    - 标签清洗 → 复读检测 → 违规检测 → 语气润色 → 分段 → 错别字模拟

12. 存储记忆
    - 这轮对话存入短期记忆
    - 关键信息存入向量库
    - 每 20 轮更新摘要

13. 输出给用户
模型切换发生在第 6 步，其他 12 步完全不变，所以对话质量不变、记忆不变、状态更新不停摆。

七、为什么这个方案能解决你的所有问题
问题 1：Gemini Flash 不输出 thinking/情绪/好感度
解决：双通道解析。Gemini Flash 不按格式输出时，语义兜底从回复内容反推情绪增量。关键词表 + 标点密度 + 用户消息共情，三层推断，保证状态不停摆。

问题 2：模型切换对话质量不变
解决：状态外置 + 每次重建 prompt。所有状态存数据库，prompt 从数据库动态拼装，与模型无关。换模型只改 body.model，其他全不变。

问题 3：简化 prompt 后质量不降
解决：强制输出协议 + 后处理管道。prompt 只保留核心要素（人设 + 状态 + 记忆 + 协议），质量靠后处理管道（复读检测、语气润色、分段）保证。

问题 4：推理模型 max_tokens 问题
解决：推理模型不设置 max_tokens + 降级 reasoning_effort 重试。不放大 max_tokens（越放大越糟），改用降级 reasoning_effort 逼迫模型压缩推理。

问题 5：不同 provider 参数差异
解决：build_provider_params 函数适配 OpenAI/Gemini/Claude/DeepSeek 各自的原生参数。

八、总结：一句话方案
状态外置 + 单次调用 + 双通道解析 + 统一后处理

状态外置：情绪/好感度/记忆存数据库，不依赖模型记忆
单次调用：主模型只调一次，不搞两步调用（省成本）
双通道解析：结构化优先（模型按格式输出时用），语义兜底（模型不按格式输出时用），默认增量保底（全失败时用）
统一后处理：所有模型输出走同一套清洗/分段/润色管道，风格趋同
这个方案的成本：与当前方案几乎相同（只多了一次语义兜底推断，本地执行，零成本），但可靠性大幅提升，且模型切换质量不变。

核心一句话：不要让模型"记住"状态，让模型"计算"增量；不要依赖模型"自觉"输出格式，用双通道解析兜底；不要让模型差异影响用户体验，用统一后处理管道抹平差异。
